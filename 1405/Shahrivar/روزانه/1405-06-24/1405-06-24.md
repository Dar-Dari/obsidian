# گزارش: سه‌شنبه ۲۴ شهریور ۱۴۰۵

**خلاصهٔ کلان روز**  
روز با تمرکز بر **انتخاب واحد دانشگاه**، **پیگیری ظرفیت دروس**، **جلسهٔ توسعهٔ ایده** و **پیشرفت فنی در پروژهٔ تشخیص انسان** سپری شد. تصمیم‌گیری‌های مهم در حوزهٔ آموزشی انجام شد و در بخش فنی، مسیر توسعهٔ مدل و محدودیت‌های سخت‌افزاری شناسایی شد.

---

### ۱. فعالیت‌های کلیدی روز

- **انتخاب واحد (۷ صبح):**
    
    - ورود به سامانه و تلاش برای انتخاب دروس مطابق فلوچارت.
    - به‌دلیل پر بودن ظرفیت درس عمومی، چیدمان دروس تغییر کرد و «کنترل خطی» جایگزین «مباحث ویژه» شد.
    - درس آقای شهشهانی در بازهٔ ۱۵:۳۰–۱۷:۳۰ انتخاب شد تا از اتلاف زمان جلوگیری شود.
- **پیگیری ظرفیت دروس:**
    
    - مذاکره با **آقای دولت‌شاهی** دربارهٔ درس نرم‌افزار با ظرفیت محدود؛ قول افزایش ظرفیت در حذف و اضافه.
    - گفت‌وگو با **آقای سبزعلی** دربارهٔ کلاس شهشهانی؛ بررسی امکان افزایش ظرفیت.
- **جلسهٔ ایده با آقای مهدوی‌کیا (۸:۳۰):**
    
    - بررسی ایدهٔ اصلی فرد مراجعه‌کننده در حوزهٔ سلامت و مسیرهای تأمین بودجه از طریق ایده‌های فرعی.
    - گفت‌وگوهای تکمیلی دربارهٔ مسیر توسعه و رفع شبهات.
- **پروژهٔ کلاه – تشخیص انسان:**
    
    - استفاده از **Edge Impulse** برای مدل تشخیص انسان روی میکروکنترلر.
    - انتخاب پروژهٔ آماده با ۲۶۰۰ نمونه و دریافت خروجی 96×96 با عملکرد مناسب.
    - تست روی **ESP32-S3**؛ خروجی سریال پرینت موفق بود اما نمایش تصویر روی TFT به‌دلیل محدودیت رم با خطا مواجه شد.
    - کاهش رزولوشن به 48×48 و تست روی S3 با حافظهٔ ۴ و ۱۶ مگ؛ مشکل همچنان پابرجا.
    - نیاز به طراحی و بهینه‌سازی جدید برای اجرای پایدار مدل.

###  چالش‌های روز

- محدودیت ظرفیت دروس کلیدی و عدم امکان انتخاب واحد مطابق فلوچارت.

- نیاز به بازطراحی اکسمپل برای مدیریت رم و اجرای هم‌زمان تصویر + تشخیص.

#include <SPI.h>

#include <TFT_eSPI.h>

#include <TJpg_Decoder.h>

#include <ArduinoWebsockets.h>

#include <WiFi.h>

  

// ۱. وارد کردن کتابخانه اختصاصی هوش مصنوعی شما که از زیپ نصب کردید

#include <Person_Detection_esp_inferencing.h>

  

#define SCREEN_WIDTH 320

#define SCREEN_HEIGHT 240

#define FONT_SIZE 2

  

const char *ssid = "ESP32CAM_to_ESP32";

const char *password = "myesp32server";

  

IPAddress local_ip(192, 168, 1, 1);

IPAddress gateway(192, 168, 1, 1);

IPAddress subnet(255, 255, 255, 0);

  

int centerX, centerY;

  

using namespace websockets;

WebsocketsServer server;

WebsocketsClient client;

  

TFT_eSPI tft = TFT_eSPI();

  

// ایجاد بافر سیاه و سفید موقت متناسب با ابعاد مدل هوش مصنوعی شما (مثلا 96x96)

#define EI_WIDTH  EI_CLASSIFIER_INPUT_WIDTH

#define EI_HEIGHT EI_CLASSIFIER_INPUT_HEIGHT

uint8_t ei_gray_buffer[EI_WIDTH * EI_HEIGHT];

  

// تابع واسط جهت تحویل پیکسل‌ها به مدل Edge Impulse

int get_signal_data(size_t offset, size_t length, float *out_ptr) {

    size_t pixel_count = length;

    size_t out_ptr_ix = 0;

    while (pixel_count > 0) {

        out_ptr[out_ptr_ix] = (float)ei_gray_buffer[offset + out_ptr_ix] / 255.0f;

        out_ptr_ix++;

        pixel_count--;

    }

    return 0;

}

  

// تابع خروجی رمزگشایی جی‌پگ (اصلاح شده برای پر کردن موازی بافر هوش مصنوعی)

bool tft_output(int16_t x, int16_t y, uint16_t w, uint16_t h, uint16_t *bitmap) {

  if (y >= tft.height()) return 0;

  // الف) نمایش مستقیم تصویر رنگی روی TFT شما بدون تغییر

  tft.pushImage(x, y, w, h, bitmap);

  // ب) تبدیل و تغییر سایز هم‌زمان پیکسل‌ها به بافر سیاه و سفید مدل هوش مصنوعی

  for (int16_t j = 0; j < h; j++) {

    int16_t current_y = y + j;

    int16_t ei_y = (current_y * EI_HEIGHT) / SCREEN_HEIGHT;

    if (ei_y >= EI_HEIGHT) continue;

  

    for (int16_t i = 0; i < w; i++) {

      int16_t current_x = x + i;

      int16_t ei_x = (current_x * EI_WIDTH) / SCREEN_WIDTH;

      if (ei_x >= EI_WIDTH) continue;

  

      // استخراج کانال‌های رنگی از فرمت RGB565 نمایشگر

      uint16_t pixel = bitmap[j * w + i];

      uint8_t r = ((pixel >> 11) & 0x1F) << 3;

      uint8_t g = ((pixel >> 5) & 0x3F) << 2;

      uint8_t b = (pixel & 0x1F) << 3;

      // محاسبه مقدار خاکستری (Grayscale)

      uint8_t gray = (uint8_t)(0.299f * r + 0.587f * g + 0.114f * b);

      ei_gray_buffer[ei_y * EI_WIDTH + ei_x] = gray;

    }

  }

  return 1;

}

  

void setup() {

  Serial.begin(115200);

  delay(1000);

  

  Serial.println("\n-------------Create ESP32 as Access Point");

  WiFi.mode(WIFI_AP);

  WiFi.setSleep(false);

  

  WiFi.softAP(ssid, password, 6);

  delay(500);

  WiFi.softAPConfig(local_ip, gateway, subnet);

  

  Serial.print("\nAP IP Address : ");

  Serial.println(WiFi.softAPIP());

  

  server.listen(8888);

  

  tft.begin();

  tft.setRotation(1);

  tft.fillScreen(TFT_BLUE);

  

  centerX = SCREEN_WIDTH / 2;

  centerY = SCREEN_HEIGHT / 2;

  

  tft.setTextColor(TFT_WHITE, TFT_BLUE);

  tft.drawCentreString("Waiting for connection", centerX, centerY - 15, FONT_SIZE);

  tft.drawCentreString("from ESP32-CAM...", centerX, centerY + 5, FONT_SIZE);

  

  tft.setSwapBytes(true);

  TJpgDec.setJpgScale(1);

  TJpgDec.setCallback(tft_output);

}

  

void loop() {

  if (server.poll()) {

    client = server.accept();

    tft.fillScreen(TFT_BLACK);

  }

  

  if (client.available()) {

    client.poll();

    WebsocketsMessage msg = client.readBlocking();

    if (msg.length() > 0) {

      // ترسیم فریم و استخراج هم‌زمان لایه‌های تصویری

      TJpgDec.drawJpg(0, 0, (const uint8_t *)msg.c_str(), msg.length());

      // ---- فرآیند آشکارسازی انسان توسط مدل FOMO ----

      signal_t signal;

      signal.total_length = EI_WIDTH * EI_HEIGHT;

      signal.get_data = &get_signal_data;

  

      ei_impulse_result_t result = { 0 };

      // فرستادن بافر به هسته پردازش شتاب‌یافته S3

      EI_IMPULSE_ERROR res = run_classifier(&signal, &result, false);

      if (res == EI_IMPULSE_OK) {

        for (size_t ix = 0; ix < result.bounding_boxes_count; ix++) {

          auto bb = result.bounding_boxes[ix];

          // فیلتر دقت: اگر درصد اطمینان بالا بود و شیء پیدا شده برچسب "person" یا "human" داشت

          if

(bb.value >= 0.70 && (strcmp(bb.label, "person") == 0 || strcmp(bb.label, "human") == 0)) {

            // بازگرداندن مختصات مدل هوش مصنوعی به مختصات دقیق ال‌سی‌دی (320x240)

            int tft_x = (bb.x * SCREEN_WIDTH) / EI_WIDTH;

            int tft_y = (bb.y * SCREEN_HEIGHT) / EI_HEIGHT;

            int tft_w = (bb.width * SCREEN_WIDTH) / EI_WIDTH;

            int tft_h = (bb.height * SCREEN_HEIGHT) / EI_HEIGHT;

            // در مدل‌های FOMO به دلیل خروجی نقطه‌ای، سایز کادر را پیش‌فرض ۳۰ پیکسل در نظر می‌گیریم

            if(tft_w == 0) tft_w = 30;

            if(tft_h == 0) tft_h = 30;

  

            // ترسیم کادر مربع قرمز دور انسان

            tft.drawRect(tft_x - (tft_w/2), tft_y - (tft_h/2), tft_w, tft_h, TFT_RED);

            // نوشتن برچسب متنی بالای سر سوژه

            tft.setTextColor(TFT_RED);

            tft.drawString("Human", tft_x - (tft_w/2), tft_y - (tft_h/2) - 12);

          }

        }

      }

    }

  }

}
